cmake_minimum_required(VERSION 3.16)
project(vs4j_count_matches C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(vs4j_count_matches SHARED count-matches-native.c)
set_target_properties(vs4j_count_matches PROPERTIES OUTPUT_NAME "vs4j_count_matches")
if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(vs4j_count_matches PRIVATE -O3)
endif ()
